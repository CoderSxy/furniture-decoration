# M4 房型 / 空间扫描 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有 M0–M3 Taro weapp 上交付房型采集：VisionKit v2 水平面测量为主路径，手填长宽高 + 最多 3 张本地相册图为降级；方案详情在量房阶段进入采集，并展示长×宽×高与简易矩形平面图。

**Architecture:** 对比文档已随本计划落盘（`docs/superpowers/specs/2026-11-03-room-scan-comparison.md`），实现时先锁候选名单再写代码，不得改推荐结论、不得改 `RoomScan`。`scanStore` 每 `planId` 至多一条扫描：新增 `upsert` 覆盖写入 `fd:scans`，以及 `getByPlanId` / `selectScanByPlanId`。能力探测与米→毫米距离为可单测纯函数；扫描页在分包 `packageScan` 内用 Taro `Camera` + 可注入的 `createVKSession`（0 npm，不引入 threejs / 重建模型）。隐私：`requirePrivacyAuthorize` + `scope.camera`，拒绝则手填。无真机时主路径标「缺真机未测」，降级 + 单测 + typecheck + `build:weapp` 必须过。

**Tech Stack:** Taro 4.3.2、React 18.3.1、TypeScript、Vite、Zustand 5、pnpm、Node `>=22.23.4 <23`、weapp；扫描用微信 VisionKit v2（0 npm）；**不新增 threejs / 重建模型 npm。**

// src/stores/scanStore.ts
import { create } from 'zustand'
import { SCANS_STORAGE_KEY } from '../constants/storageKeys'
import type { RoomScan } from '../types/domain'
import { readJSON, writeJSON } from '../utils/storage'

function parsePhotoPaths(input: unknown): string[] {
  if (!Array.isArray(input)) return []
  return input.filter((item): item is string => typeof item === 'string')
}

function parseScan(key: string, input: unknown): RoomScan | null {
  if (input == null || typeof input !== 'object') return null
  const row = input as Record<string, unknown>
  if (typeof row.planId !== 'string' || row.planId.length === 0 || row.planId !== key) return null
  if (row.source !== 'visionkit-v2' && row.source !== 'manual') return null
  if (typeof row.widthMm !== 'number' || !Number.isFinite(row.widthMm)) return null
  if (typeof row.depthMm !== 'number' || !Number.isFinite(row.depthMm)) return null

import type { CreateVkSession, VkSessionHandle } from '../types/vk'

export const MIN_VISIONKIT_SDK_VERSION = '2.22.0'

export type ScanCapability = 'visionkit-v2' | 'manual'

export type ProbeResult =
  | { capability: 'manual'; session: null }
  | { capability: 'visionkit-v2'; session: VkSessionHandle }

export type ScanCapabilityDeps = {
  getSdkVersion: () => string
  createVKSession?: CreateVkSession
}

export function parseSdkVersion(version: string): [number, number, number] {
  const parts = String(version ?? '')
    .split('.')
    .map((part) => Number.parseInt(part, 10))
  const n0 = Number.isFinite(parts[0]) ? parts[0] : 0
  const n1 = Number.isFinite(parts[1]) ? parts[1] : 0
  const n2 = Number.isFinite(parts[2]) ? parts[2] : 0
  return [n0, n1, n2]
}

export function compareSdkVersion(a: string, b: string): number {
  const left = parseSdkVersion(a)
  const right = parseSdkVersion(b)
  for (let i = 0; i < 3; i += 1) {
    if (left[i] !== right[i]) return left[i] - right[i]
  }
  return 0
}

export function isVisionKitV2SdkSupported(sdkVersion: string): boolean {
  return compareSdkVersion(sdkVersion, MIN_VISIONKIT_SDK_VERSION) >= 0
}

export function isVkStartError(status: unknown): boolean {
  if (status == null || status === 0 || status === '0' || status === 'success') return false
  return true
}

export function probeScanCapability(deps: ScanCapabilityDeps): Promise<ProbeResult> {
  return new Promise((resolve) => {
    if (!isVisionKitV2SdkSupported(deps.getSdkVersion()) || typeof deps.createVKSession !== 'function') {
      resolve({ capability: 'manual', session: null })
      return
    }
    let session: VkSessionHandle
    try {
      session = deps.createVKSession({ version: 'v2', track: { plane: { mode: 1 } } })
    } catch {
      resolve({ capability: 'manual', session: null })
      return
    }
    try {
      session.start((status) => {
        if (isVkStartError(status)) {
          try {
            session.destroy()
          } catch {
            // ignore destroy errors on fallback
          }
          resolve({ capability: 'manual', session: null })
          return
        }
        resolve({ capability: 'visionkit-v2', session })
      })
    } catch {
      try {
        session.destroy()
      } catch {
        // ignore destroy errors on fallback
      }
      resolve({ capability: 'manual', session: null })
    }
  })
}

// src/utils/scanMeasure.ts
import type { RoomScan } from '../types/domain'

export const DEFAULT_ROOM_HEIGHT_MM = 2800
export const MAX_SCAN_PHOTOS = 3

export type Vec3 = { x: number; y: number; z: number }

export function metersToMm(meters: number): number {
  return Math.round(meters * 1000)
}

export function translationFromHitTransform(transform: ArrayLike<number>): Vec3 | null {
  if (transform.length < 16) return null
  const x = Number(transform[12])
  const y = Number(transform[13])
  const z = Number(transform[14])
  if (!Number.isFinite(x) || !Number.isFinite(y) || !Number.isFinite(z)) return null
  return { x, y, z }
}

export function distanceMeters(a: Vec3, b: Vec3): number {
  const dx = a.x - b.x
  const dy = a.y - b.y
  const dz = a.z - b.z
  return Math.sqrt(dx * dx + dy * dy + dz * dz)
}

export function distanceMmBetweenTransforms(
  a: ArrayLike<number>,
  b: ArrayLike<number>,
): number | null {
  const pa = translationFromHitTransform(a)
  const pb = translationFromHitTransform(b)
  if (!pa || !pb) return null
  const mm = metersToMm(distanceMeters(pa, pb))
  if (!(mm > 0)) return null
  return mm
}

export type ManualScanDraft = {
  planId: string
  widthText: string
  depthText: string
  heightText: string
  photoPaths: string[]
}

export type ManualScanError =
  | 'invalid-plan'
  | 'invalid-width'
  | 'invalid-depth'
  | 'invalid-height'
  | 'invalid-photos'

export const MANUAL_SCAN_ERRORS: Record<ManualScanError, string> = {
  'invalid-plan': '缺少方案，无法采集',
  'invalid-width': '长度必须大于 0',
  'invalid-depth': '宽度必须大于 0',
  'invalid-height': '层高必须大于 0',
  'invalid-photos': '参考图最多 3 张',
}

export function parsePositiveMm(text: string): number | null {
  const value = Number(String(text).trim())
  if (!Number.isFinite(value) || value <= 0) return null
  return value
}

export function validateManualScan(draft: ManualScanDraft): ManualScanError | 'ok' {
  if (typeof draft.planId !== 'string' || draft.planId.length === 0) return 'invalid-plan'
  if (parsePositiveMm(draft.widthText) == null) return 'invalid-width'
  if (parsePositiveMm(draft.depthText) == null) return 'invalid-depth'
  if (parsePositiveMm(draft.heightText) == null) return 'invalid-height'
  if (!Array.isArray(draft.photoPaths) || draft.photoPaths.length > MAX_SCAN_PHOTOS) {
    return 'invalid-photos'
  }
  if (draft.photoPaths.some((item) => typeof item !== 'string' || item.length === 0)) {
    return 'invalid-photos'
  }
  return 'ok'
}

export function appendPhotoPaths(current: string[], incoming: string[]): string[] {
  const next = [...current]
  for (const path of incoming) {
    if (typeof path !== 'string' || path.length === 0) continue
    if (next.includes(path)) continue
    if (next.length >= MAX_SCAN_PHOTOS) break
    next.push(path)
  }
  return next
}

export function buildManualScan(draft: ManualScanDraft, capturedAt: string): RoomScan {
  const widthMm = parsePositiveMm(draft.widthText)
  const depthMm = parsePositiveMm(draft.depthText)
  const heightMm = parsePositiveMm(draft.heightText)
  return {
    planId: draft.planId,
    source: 'manual',
    widthMm: widthMm ?? 0,
    depthMm: depthMm ?? 0,
    heightMm: heightMm ?? 0,
    photoPaths: draft.photoPaths.slice(0, MAX_SCAN_PHOTOS),
    capturedAt,
  }
}

export function buildVisionKitScan(input: {
  planId: string
  widthMm: number
  depthMm: number
  heightMm: number
  capturedAt: string
}): RoomScan {
  return {
    planId: input.planId,
    source: 'visionkit-v2',
    widthMm: input.widthMm,
    depthMm: input.depthMm,
    heightMm: input.heightMm,
    photoPaths: [],
    capturedAt: input.capturedAt,
  }
}

export type MeasurePhase = 'width-a' | 'width-b' | 'depth-a' | 'depth-b' | 'height'

export type MeasureState = {
  phase: MeasurePhase
  widthMm: number | null
  depthMm: number | null
  heightMm: number
  pending: ArrayLike<number> | null
}

export function createMeasureState(heightMm = DEFAULT_ROOM_HEIGHT_MM): MeasureState {
  return {
    phase: 'width-a',
    widthMm: null,
    depthMm: null,
    heightMm,
    pending: null,
  }
}

export function measureHint(phase: MeasurePhase): string {
  if (phase === 'width-a') return '点击地面第一点，测量长度'
  if (phase === 'width-b') return '再点击地面第二点，完成长度'
  if (phase === 'depth-a') return '点击地面第一点，测量宽度'
  if (phase === 'depth-b') return '再点击地面第二点，完成宽度'
  return '确认层高后保存'
}

export function canSaveMeasure(state: MeasureState): boolean {
  return (
    state.phase === 'height' &&
    state.widthMm != null &&
    state.widthMm > 0 &&
    state.depthMm != null &&
    state.depthMm > 0 &&
    state.heightMm > 0
  )
}

export function applyHit(
  state: MeasureState,
  transform: ArrayLike<number>,
): { state: MeasureState; error?: 'no-distance' } {
  if (state.phase === 'height') return { state }
  if (state.phase === 'width-a' || state.phase === 'depth-a') {
    return {
      state: {
        ...state,
        pending: transform,
        phase: state.phase === 'width-a' ? 'width-b' : 'depth-b',
      },
    }
  }
  if (state.pending == null) return { state, error: 'no-distance' }
  const mm = distanceMmBetweenTransforms(state.pending, transform)
  if (mm == null) return { state, error: 'no-distance' }
  if (state.phase === 'width-b') {
    return {
      state: {
        ...state,
        widthMm: mm,
        pending: null,
        phase: 'depth-a',
      },
    }
  }
  return {
    state: {
      ...state,
      depthMm: mm,
      pending: null,
      phase: 'height',
    },
  }
}

export type PageRect = { left: number; top: number; width: number; height: number }

export function normalizePageTap(
  pageX: number,
  pageY: number,
  rect: PageRect,
): { x: number; y: number } | null {
  if (!(rect.width > 0) || !(rect.height > 0)) return null
  const x = (pageX - rect.left) / rect.width
  const y = (pageY - rect.top) / rect.height
  return {
    x: Math.min(1, Math.max(0, x)),
    y: Math.min(1, Math.max(0, y)),
  }
}

// src/utils/weappScanBridge.ts
import Taro from '@tarojs/taro'
import type { CreateVkSession } from '../types/vk'
import type { PrivacyBridge } from './scanPrivacy'

export function readSdkVersion(): string {
  try {
    const info = Taro.getSystemInfoSync()
    return typeof info.SDKVersion === 'string' && info.SDKVersion.length > 0
      ? info.SDKVersion
      : '0.0.0'
  } catch {
    return '0.0.0'
  }
}

export function getCreateVkSession(): CreateVkSession | undefined {
  const taro = Taro as unknown as { createVKSession?: CreateVkSession }
  if (typeof taro.createVKSession === 'function') {
    return (options) => taro.createVKSession!(options)
  }
  const wxObj = (globalThis as { wx?: { createVKSession?: CreateVkSession } }).wx
  if (wxObj && typeof wxObj.createVKSession === 'function') {
    return (options) => wxObj.createVKSession!(options)
  }
  return undefined
}

export function createWeappPrivacyBridge(): PrivacyBridge {
  return {
    requirePrivacyAuthorize: () =>
      new Promise((resolve, reject) => {
        const api = Taro as unknown as {
          requirePrivacyAuthorize?: (option: { success: () => void; fail: () => void }) => void
        }
        if (typeof api.requirePrivacyAuthorize !== 'function') {
          resolve()
          return
        }
        api.requirePrivacyAuthorize({
          success: () => resolve(),
          fail: () => reject(new Error('privacy-denied')),
        })
      }),
    authorizeCamera: () => Taro.authorize({ scope: 'scope.camera' }).then(() => undefined),
  }
}

import { Camera, Input, Text, View } from '@tarojs/components'
import Taro from '@tarojs/taro'
import { useState } from 'react'
import { useScanStore } from '../../../stores/scanStore'
import type { VkSessionHandle } from '../../../types/vk'
import {
  applyHit,
  buildVisionKitScan,
  canSaveMeasure,
  createMeasureState,
  DEFAULT_ROOM_HEIGHT_MM,
  measureHint,
  normalizePageTap,
  parsePositiveMm,
} from '../../../utils/scanMeasure'

export type MeasureViewProps = {
  planId: string
  session: VkSessionHandle
  onSaved: () => void
  onFallback: () => void
}

export function MeasureView({ planId, session, onSaved, onFallback }: MeasureViewProps) {
  const upsert = useScanStore((state) => state.upsert)
  const [state, setState] = useState(() => createMeasureState())
  const [heightText, setHeightText] = useState(String(DEFAULT_ROOM_HEIGHT_MM))

  const onTap = (event: {
    detail?: { x?: number; y?: number }
    changedTouches?: Array<{ pageX?: number; clientX?: number; pageY?: number; clientY?: number }>
  }) => {
    const touch = event.changedTouches?.[0]
    const pageX = touch?.pageX ?? touch?.clientX ?? event.detail?.x
    const pageY = touch?.pageY ?? touch?.clientY ?? event.detail?.y
    if (pageX == null || pageY == null) return
    Taro.createSelectorQuery()
      .select('.scan-stage')
      .boundingClientRect()
      .exec((res) => {
        const rect = res?.[0] as { left: number; top: number; width: number; height: number } | undefined
        if (!rect) return
        const tap = normalizePageTap(pageX, pageY, rect)
        if (!tap) return
        const hits = session.hitTest(tap.x, tap.y)
        const transform = hits[0]?.transform
        if (!transform) {
          Taro.showToast({ title: '未识别到地面，请对准地面再点', icon: 'none' })
          return
        }
        setState((current) => {
          const next = applyHit(current, transform)
          if (next.error === 'no-distance') {
            Taro.showToast({ title: '未识别到地面，请对准地面再点', icon: 'none' })
            return current
          }
          return next.state
        })
      })
  }

  const save = () => {
    const heightMm = parsePositiveMm(heightText)
    if (heightMm == null) {
      Taro.showToast({ title: '层高必须大于 0', icon: 'none' })
      return
    }
    const ready = { ...state, heightMm }
    if (!canSaveMeasure(ready) || ready.widthMm == null || ready.depthMm == null) {
      Taro.showToast({ title: '请先点完地面两对点', icon: 'none' })
      return
    }
    upsert(
      buildVisionKitScan({
        planId,
        widthMm: ready.widthMm,
        depthMm: ready.depthMm,
        heightMm,
        capturedAt: new Date().toISOString(),
      }),
    )
    Taro.showToast({ title: '已保存房型', icon: 'success' })
    onSaved()
  }

  return (
    <View className='measure'>
      <View className='scan-stage' onClick={(event) => onTap(event)}>
        <Camera
          className='scan-camera'
          devicePosition='back'
          flash='off'
          onError={() => onFallback()}
        />
        <View className='scan-hit' />
      </View>
      <Text className='measure__hint'>{measureHint(state.phase)}</Text>
      <Text className='measure__meta'>
        请将手机对准地面并左右平移完成初始化。长度 {state.widthMm ?? '—'} mm · 宽度{' '}
        {state.depthMm ?? '—'} mm
      </Text>
      <Input
        className='manual__input'
        type='number'
        value={heightText}
        placeholder='层高（mm）'
        onInput={(event) => {
          const next = event.detail.value
          setHeightText(next)
          const heightMm = parsePositiveMm(next) ?? DEFAULT_ROOM_HEIGHT_MM
          setState((current) => ({ ...current, heightMm }))
        }}
      />
      <View className='manual__submit' onClick={save}>
        <Text className='manual__submit-text'>保存测量</Text>
      </View>
      <View className='manual__secondary' onClick={onFallback}>
        <Text className='manual__secondary-text'>改用手填</Text>
      </View>
    </View>
  )
}

import Taro, { useRouter } from '@tarojs/taro'
import { useEffect, useState } from 'react'
import type { ProbeResult } from '../../../utils/scanCapability'
import { probeScanCapability } from '../../../utils/scanCapability'
import { ensureCameraPrivacy } from '../../../utils/scanPrivacy'
import {
  createWeappPrivacyBridge,
  getCreateVkSession,
  readSdkVersion,
} from '../../../utils/weappScanBridge'
import { ManualForm } from './manual-form'
import { MeasureView } from './measure-view'
import './index.scss'

export default function ScanPage() {
  const router = useRouter()
  const planId = router.params.planId ?? ''
  const [mode, setMode] = useState<'loading' | 'manual' | 'visionkit'>('loading')
  const [session, setSession] = useState<ProbeResult['session']>(null)

  useEffect(() => {
    if (!planId) return
    let cancelled = false
    let active: ProbeResult['session'] = null
    const run = async () => {
      const result = await probeScanCapability({
        getSdkVersion: readSdkVersion,
        createVKSession: getCreateVkSession(),
      })
      active = result.session
      if (cancelled) {
        active?.destroy()
        return

      if (result.capability !== 'visionkit-v2' || result.session == null) {
        setMode('manual')
        return
      }
      const allowed = await ensureCameraPrivacy(createWeappPrivacyBridge())
      if (cancelled) {
        result.session.destroy()
        return
      }
      if (!allowed) {
        result.session.destroy()
        active = null
        setMode('manual')
        return
      }
      setSession(result.session)
      setMode('visionkit')
    }
    void run()
    return () => {
      cancelled = true
      active?.destroy()
    }

      <View className='page'>
        <Text className='page__title'>缺少方案，无法采集</Text>
      </View>
    )
  }

  if (mode === 'loading') {
    return (
      <View className='page'>
        <Text className='page__duty'>正在检测测量能力</Text>
      </View>
    )
  }

  if (mode === 'visionkit' && session) {
    return (
      <View className='page page--measure'>
        <MeasureView
          planId={planId}
          session={session}
          onSaved={() => {
            session.destroy()
            Taro.navigateBack()
          }}
          onFallback={() => {
            session.destroy()
            setSession(null)
            setMode('manual')
          }}
        />
      </View>
    )
  }

  return (
    <View className='page'>
      <ManualForm
        planId={planId}
        onSaved={() => {
          Taro.navigateBack()
        }}
      />
    </View>

import Taro, { useRouter } from '@tarojs/taro'
import { ProductCard } from '../../components/product-card'
import { RoomFloorplan } from '../../components/room-floorplan'

import { useBudgetStore } from '../../stores/budgetStore'
import { useCatalogStore } from '../../stores/catalogStore'
import { selectPlanById, usePlanStore } from '../../stores/planStore'
import { selectScanByPlanId, useScanStore } from '../../stores/scanStore'

import { productDetailUrl, scanPageUrl } from '../../utils/navigation'
import { advanceBlockedReason, canAdvanceStage, formatPlanArea } from '../../utils/plans'
import './detail.scss'

export default function PlanDetailPage() {
  const router = useRouter()
  const planId = router.params.planId ?? ''
  const plan = usePlanStore((state) => selectPlanById(state, planId))
  const scan = useScanStore((state) => selectScanByPlanId(state, planId))
  const removeProductFromPlan = usePlanStore((state) => state.removeProductFromPlan)
  const advanceStage = usePlanStore((state) => state.advanceStage)

      </View>
    )

  const planProducts: Product[] = plan.productIds

    .filter((item): item is Product => item != null)
  const blocked = advanceBlockedReason(plan)
  const canAdvance = canAdvanceStage(plan)

      <Text className='plan-title'>{plan.title}</Text>

      </Text>

          const current = index === currentIndex
          const done = index < currentIndex

            : done
              ? 'plan-timeline__item plan-timeline__item--done'
              : 'plan-timeline__item'

        </View>
      ) : null}
      {scan ? (
        <View className='plan-scan'>
          <Text className='plan-section'>房型尺寸</Text>
          <Text className='plan-scan__size'>

          </Text>
          <Text className='plan-scan__source'>{SCAN_SOURCE_LABELS[scan.source]}</Text>
          <RoomFloorplan widthMm={scan.widthMm} depthMm={scan.depthMm} />
        </View>
      ) : null}
      <Text className='plan-section'>已选商品</Text>
      {planProducts.length === 0 ? (
        <View
          className='plan-empty-products'

          }}
        >
          <Text className='plan-empty-products__text'>还没有商品，去选品加入</Text>

              }}
            />
            <View className='plan-product__remove' onClick={() => onRemove(product.id)}>
              <Text className='plan-product__remove-text'>移除</Text>
            </View>

}
