// src/utils/price.ts
import type { PriceRange } from '../types/domain'

export function priceFenToYuan(priceFen: number): number {
  return Math.round(priceFen / 100)
}

export function formatPriceYuan(priceFen: number): string {
  return `¥${priceFenToYuan(priceFen)}`
}

export function matchesPriceRange(priceFen: number, range: PriceRange): boolean {
  if (range === 'all') return true
  const yuan = priceFenToYuan(priceFen)
  if (range === 'lt1000') return yuan < 1000
  if (range === '1000to3000') return yuan >= 1000 && yuan < 3000
  if (range === '3000to8000') return yuan >= 3000 && yuan < 8000
  return yuan >= 8000
}
```

// src/utils/filterProducts.ts
import {
  CATEGORY_LABELS,
  PRICE_RANGE_LABELS,
  ROOM_LABELS,
  STYLE_LABELS,
} from '../constants/labels'
import type { CatalogFilters, Product } from '../types/domain'
import { matchesPriceRange } from './price'

export function filterProducts(products: Product[], filters: CatalogFilters): Product[] {
  return products.filter((product) => {
    if (filters.category !== 'all' && product.category !== filters.category) return false
    if (filters.room !== 'all' && !product.rooms.includes(filters.room)) return false
    if (filters.style !== 'all' && product.style !== filters.style) return false
    if (!matchesPriceRange(product.priceFen, filters.priceRange)) return false
    return true
  })
}

export function hasActiveFilters(filters: CatalogFilters): boolean {
  return (
    filters.category !== 'all' ||
    filters.room !== 'all' ||
    filters.style !== 'all' ||
    filters.priceRange !== 'all'
  )
}

export function summarizeFilters(filters: CatalogFilters): string {
  if (!hasActiveFilters(filters)) return '全部商品'
  const parts: string[] = []
  if (filters.category !== 'all') parts.push(CATEGORY_LABELS[filters.category])
  if (filters.room !== 'all') parts.push(ROOM_LABELS[filters.room])
  if (filters.style !== 'all') parts.push(STYLE_LABELS[filters.style])
  if (filters.priceRange !== 'all') parts.push(PRICE_RANGE_LABELS[filters.priceRange])
  return parts.join(' · ')
}
```

// src/utils/storage.ts
export type KeyValueStorage = {
  getItem: (key: string) => string | null
  setItem: (key: string, value: string) => void
}

let storage: KeyValueStorage = {
  getItem: () => null,
  setItem: () => undefined,
}

export function configureStorage(next: KeyValueStorage): void {
  storage = next
}

export function createMemoryStorage(
  initial: Record<string, string> = {},
): KeyValueStorage {
  const map = new Map<string, string>(Object.entries(initial))
  return {
    getItem: (key) => (map.has(key) ? map.get(key)! : null),
    setItem: (key, value) => {
      map.set(key, value)
    },
  }
}

export function readJSON<T>(key: string, fallback: T): T {
  const raw = storage.getItem(key)
  if (raw == null || raw === '') return fallback
  try {
    return JSON.parse(raw) as T
  } catch {
    return fallback
  }
}

export function writeJSON(key: string, value: unknown): void {
  storage.setItem(key, JSON.stringify(value))
}
```

import { create } from 'zustand'
import { PRODUCTS } from '../mocks/products'
import type { CatalogFilters, Product } from '../types/domain'

export const DEFAULT_FILTERS: CatalogFilters = {
  category: 'all',
  room: 'all',
  style: 'all',
  priceRange: 'all',
}

export type CatalogState = {
  products: Product[]
  filters: CatalogFilters
  setFilters: (patch: Partial<CatalogFilters>) => void
  clearFilters: () => void
}

export const useCatalogStore = create<CatalogState>((set) => ({

  clearFilters: () => set({ filters: { ...DEFAULT_FILTERS } }),
}))

export function selectFilteredProducts(state: CatalogState): Product[] {
  return filterProducts(state.products, state.filters)
}

import { FAVORITES_STORAGE_KEY } from '../constants/storageKeys'
import type { FavoriteRecord } from '../types/domain'
import { readJSON, writeJSON } from '../utils/storage'

export function parseFavoriteRecords(input: unknown): FavoriteRecord[] {
  if (!Array.isArray(input)) return []
  const records: FavoriteRecord[] = []
  for (const row of input) {
    if (row == null || typeof row !== 'object') continue
    const productId = (row as { productId?: unknown }).productId
    const savedAt = (row as { savedAt?: unknown }).savedAt
    if (typeof productId !== 'string' || productId.length === 0) continue
    if (typeof savedAt !== 'string' || savedAt.length === 0) continue
    records.push({ productId, savedAt })
  }
  return records
}

export type FavoriteState = {
  items: FavoriteRecord[]
  hydrate: () => void
  toggleFavorite: (productId: string) => void
  isFavorite: (productId: string) => boolean
}

export const useFavoriteStore = create<FavoriteState>((set, get) => ({
  items: [],
  hydrate: () => {
    set({ items: parseFavoriteRecords(readJSON(FAVORITES_STORAGE_KEY, [])) })
  },
  toggleFavorite: (productId: string) => {
    const current = get().items
    const exists = current.some((item) => item.productId === productId)
    const items = exists
      ? current.filter((item) => item.productId !== productId)
      : [...current, { productId, savedAt: new Date().toISOString() }]
    writeJSON(FAVORITES_STORAGE_KEY, items)
    set({ items })
  },
  isFavorite: (productId: string) => get().items.some((item) => item.productId === productId),
}))
```

export type FilterBarProps = {
  filters: CatalogFilters
  onChange: (patch: Partial<CatalogFilters>) => void
  onClear: () => void

const ALL_LABEL = '全部'

function ChipRow<T extends string>({
  options,
  labelOf,
  value,
  onSelect,
}: {
  options: readonly T[]
  labelOf: (option: T) => string
  value: T
  onSelect: (option: T) => void
}) {
  return (
    <View className='filter-bar__row'>
      {options.map((option) => {
        const active = option === value
        return (
          <View
            key={option}
            className={active ? 'filter-bar__chip filter-bar__chip--active' : 'filter-bar__chip'}
            onClick={() => onSelect(option)}
          >
            <Text
              className={
                active ? 'filter-bar__chip-text filter-bar__chip-text--active' : 'filter-bar__chip-text'
              }
            >
              {labelOf(option)}
            </Text>
          </View>
        )
      })}
    </View>
  )
}

export function FilterBar({ filters, onChange, onClear }: FilterBarProps) {
  return (
    <View className='filter-bar'>
      <Text className='filter-bar__legend'>品类</Text>
      <ChipRow
        options={CATEGORY_FILTER_OPTIONS}
        value={filters.category}
        labelOf={(option) => (option === 'all' ? ALL_LABEL : CATEGORY_LABELS[option as ProductCategory])}

// src/pages/mine/index.tsx
import { Text, View } from '@tarojs/components'
import Taro from '@tarojs/taro'
import { ProductCard } from '../../components/product-card'
import { useCatalogStore } from '../../stores/catalogStore'
import { useFavoriteStore } from '../../stores/favoriteStore'
import type { Product } from '../../types/domain'
import { productDetailUrl } from '../../utils/navigation'
import './index.scss'

export default function MinePage() {
  const items = useFavoriteStore((state) => state.items)
  const products = useCatalogStore((state) => state.products)
  const favoriteProducts: Product[] = [...items]
    .sort((left, right) => right.savedAt.localeCompare(left.savedAt))
    .map((record) => products.find((product) => product.id === record.productId))
    .filter((product): product is Product => product != null)

  return (
    <View className='page'>
      <Text className='page__title'>收藏</Text>
      {favoriteProducts.length === 0 ? (
        <View
          className='mine-empty'
          onClick={() => {
            Taro.switchTab({ url: '/pages/catalog/index' })
          }}
        >
          <Text className='mine-empty__text'>还没有收藏，去选品看看</Text>
        </View>
      ) : (
        favoriteProducts.map((product) => (
          <ProductCard
            key={product.id}
            product={product}
            onTap={(productId) => {
              Taro.navigateTo({ url: productDetailUrl(productId) })
            }}
          />
        ))
      )}
    </View>
  )
}

// src/app.ts
import { PropsWithChildren } from 'react'
import Taro from '@tarojs/taro'
import { useFavoriteStore } from './stores/favoriteStore'
import { configureStorage } from './utils/storage'

import './app.scss'

configureStorage({
  getItem: (key) => {
    try {
      const value = Taro.getStorageSync(key)
      if (value === '' || value == null) return null
      return typeof value === 'string' ? value : JSON.stringify(value)
    } catch {
      return null
    }
  },
  setItem: (key, value) => {
    Taro.setStorageSync(key, value)
  },
})

useFavoriteStore.getState().hydrate()

  return children
}
