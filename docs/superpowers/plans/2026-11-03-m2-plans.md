# M2 装修方案流程 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有 M0+M1 Taro weapp 壳上交付装修方案：列表卡片、新建、当前方案标记、四阶段只前进状态机、方案详情商品增删，以及选品详情的「加入当前方案」。

**Architecture:** 不重做脚手架，不改 `Product` / `Plan` / `PlanStage` 类型。阶段文案与顺序进已有 `labels.ts`。`planStore` 用已有 `configureStorage` 把 `{ plans, activePlanId }` 写入单一 Taro key `fd:plans`。列表继续用 `pages/plans/index`（含新建表单），详情新增主包页 `pages/plans/detail`（query 只传 `planId`）。点卡片 `navigateTo` 详情且**不**改 `activePlanId`；「设为当前」才改。选品详情出现「加入当前方案」；无 `activePlanId` 时按钮改为「请先新建方案」并 `switchTab` 到方案 Tab。加入商品**不**自动推进阶段。M2 不写预算、不写扫描/AI 入口。

**Tech Stack:** Taro 4.3.2、React 18.3.1、TypeScript、Vite、Zustand 5、pnpm、Node.js `>=22.23.4 <23`、目标端仅 weapp。不新增 npm 依赖。

```ts
// src/utils/plans.ts
import { ROOM_CREATE_OPTIONS, STAGE_ORDER } from '../constants/labels'
import type { Plan, PlanStage, RoomType } from '../types/domain'

export type CreatePlanInput = {
  title: string
  roomType: RoomType
  areaM2: number
}

export type CreatePlanError = 'invalid-title' | 'invalid-room' | 'invalid-area'

export function validateCreatePlan(input: {
  title: string
  roomType: string
  areaM2: number
}): 'ok' | CreatePlanError {
  if (input.title.trim().length === 0) return 'invalid-title'
  if (!(ROOM_CREATE_OPTIONS as readonly string[]).includes(input.roomType)) return 'invalid-room'
  if (!Number.isFinite(input.areaM2) || input.areaM2 <= 0) return 'invalid-area'
  return 'ok'
}

export function nextStage(stage: PlanStage): PlanStage | null {
  const index = STAGE_ORDER.indexOf(stage)
  if (index < 0 || index >= STAGE_ORDER.length - 1) return null
  return STAGE_ORDER[index + 1]
}

export function advanceBlockedReason(
  plan: Pick<Plan, 'stage' | 'productIds'>,
): string | null {
  if (nextStage(plan.stage) == null) return '已验收'
  if (plan.stage === 'measure' && plan.productIds.length < 1) return '请先去选品'
  return null
```

```ts
// src/stores/planStore.ts
import { create } from 'zustand'
import { ROOM_CREATE_OPTIONS, STAGE_ORDER } from '../constants/labels'
import { PLANS_STORAGE_KEY } from '../constants/storageKeys'
import type { Plan, PlanStage, RoomType } from '../types/domain'
import {
  canAdvanceStage,
  createPlanId,
  nextStage,
  validateCreatePlan,
  type CreatePlanError,
  type CreatePlanInput,
} from '../utils/plans'
import { readJSON, writeJSON } from '../utils/storage'

export type PersistedPlans = {
  plans: Plan[]
  activePlanId: string | null
}

export type AddProductResult = 'added' | 'duplicate' | 'no-active-plan'

export type CreatePlanResult =
  | { ok: true; plan: Plan }
  | { ok: false; reason: CreatePlanError }

function isRoomType(value: unknown): value is RoomType {
  return typeof value === 'string' && (ROOM_CREATE_OPTIONS as readonly string[]).includes(value)
}

function isPlanStage(value: unknown): value is PlanStage {
  return typeof value === 'string' && (STAGE_ORDER as readonly string[]).includes(value)
}

function parseProductIds(input: unknown): string[] {
  if (!Array.isArray(input)) return []
  const ids: string[] = []
  for (const item of input) {
    if (typeof item !== 'string' || item.length === 0) continue
    if (!ids.includes(item)) ids.push(item)
  }
  return ids
}

function parsePlan(input: unknown): Plan | null {
  if (input == null || typeof input !== 'object') return null
  const row = input as Record<string, unknown>
  if (typeof row.id !== 'string' || row.id.length === 0) return null
  if (typeof row.title !== 'string' || row.title.trim().length === 0) return null
  if (!isRoomType(row.roomType)) return null
  if (typeof row.areaM2 !== 'number' || !Number.isFinite(row.areaM2) || row.areaM2 <= 0) return null
  if (!isPlanStage(row.stage)) return null
  if (typeof row.createdAt !== 'string' || row.createdAt.length === 0) return null
  if (typeof row.updatedAt !== 'string' || row.updatedAt.length === 0) return null
  const productIds = parseProductIds(row.productIds)
  const plan: Plan = {
    id: row.id,
    title: row.title,
    roomType: row.roomType,
    areaM2: row.areaM2,
    stage: row.stage,
    productIds,
    createdAt: row.createdAt,
    updatedAt: row.updatedAt,
  }
  if (typeof row.coverProductId === 'string' && row.coverProductId.length > 0) {
    plan.coverProductId = row.coverProductId
  }
  return plan
}

export function parsePersistedPlans(input: unknown): PersistedPlans {
  if (input == null || typeof input !== 'object' || Array.isArray(input)) {
    return { plans: [], activePlanId: null }
  }
  const row = input as { plans?: unknown; activePlanId?: unknown }
  const plans = Array.isArray(row.plans)
    ? row.plans.map(parsePlan).filter((item): item is Plan => item != null)
    : []
  const activePlanId =
    typeof row.activePlanId === 'string' && plans.some((item) => item.id === row.activePlanId)
      ? row.activePlanId
      : null
  return { plans, activePlanId }
}

function persist(plans: Plan[], activePlanId: string | null): void {
  const payload: PersistedPlans = { plans, activePlanId }
  writeJSON(PLANS_STORAGE_KEY, payload)
}

export type PlanState = {
  plans: Plan[]
  activePlanId: string | null
  hydrate: () => void
  createPlan: (input: CreatePlanInput) => CreatePlanResult
  setActivePlanId: (planId: string) => void
  addProductToActivePlan: (productId: string) => AddProductResult
  removeProductFromPlan: (planId: string, productId: string) => void
  advanceStage: (planId: string) => boolean
}

export const usePlanStore = create<PlanState>((set, get) => ({
  plans: [],
  activePlanId: null,
  hydrate: () => {
    const parsed = parsePersistedPlans(readJSON<unknown>(PLANS_STORAGE_KEY, null))
    set(parsed)
  },
  createPlan: (input) => {
    const reason = validateCreatePlan(input)
    if (reason !== 'ok') return { ok: false, reason }
    const now = new Date().toISOString()
    const plan: Plan = {
      id: createPlanId(),
      title: input.title.trim(),
      roomType: input.roomType,
      areaM2: input.areaM2,
      stage: 'measure',
      productIds: [],
      createdAt: now,
      updatedAt: now,
    }
    const plans = [...get().plans, plan]
    persist(plans, plan.id)
    set({ plans, activePlanId: plan.id })
    return { ok: true, plan }
  },
  setActivePlanId: (planId) => {
    if (!get().plans.some((item) => item.id === planId)) return
    persist(get().plans, planId)
    set({ activePlanId: planId })
  },
  addProductToActivePlan: (productId) => {
    const { plans, activePlanId } = get()
    if (activePlanId == null) return 'no-active-plan'
    const current = plans.find((item) => item.id === activePlanId)
    if (!current) return 'no-active-plan'
    if (current.productIds.includes(productId)) return 'duplicate'
    const productIds = [...current.productIds, productId]
    const coverProductId = current.coverProductId ?? productId
    const next = plans.map((item) =>
      item.id === activePlanId
        ? { ...item, productIds, coverProductId, updatedAt: new Date().toISOString() }
        : item,
    )
    persist(next, activePlanId)
    set({ plans: next })
    return 'added'
  },
  removeProductFromPlan: (planId, productId) => {
    const next = get().plans.map((item) => {
      if (item.id !== planId) return item
      const productIds = item.productIds.filter((id) => id !== productId)
      const coverProductId =
        item.coverProductId === productId ? productIds[0] : item.coverProductId
      return { ...item, productIds, coverProductId, updatedAt: new Date().toISOString() }
    })
    persist(next, get().activePlanId)
    set({ plans: next })
  },
  advanceStage: (planId) => {
    const current = get().plans.find((item) => item.id === planId)
    if (!current || !canAdvanceStage(current)) return false
    const stage = nextStage(current.stage)
    if (stage == null) return false
    const next = get().plans.map((item) =>
      item.id === planId ? { ...item, stage, updatedAt: new Date().toISOString() } : item,
    )
    persist(next, get().activePlanId)
    set({ plans: next })
    return true
  },
}))

export function selectActivePlan(state: PlanState): Plan | undefined {
  if (state.activePlanId == null) return undefined
  return state.plans.find((item) => item.id === state.activePlanId)
}

export function selectPlanById(state: PlanState, planId: string): Plan | undefined {
  return state.plans.find((item) => item.id === planId)
}
```

```tsx
import Taro, { useRouter } from '@tarojs/taro'
import { ProductCard } from '../../components/product-card'
import { ROOM_LABELS, STAGE_LABELS, STAGE_ORDER } from '../../constants/labels'

import { advanceBlockedReason, canAdvanceStage, formatPlanArea } from '../../utils/plans'

  const router = useRouter()
  const planId = router.params.planId ?? ''
  const plan = usePlanStore((state) => selectPlanById(state, planId))

        <Text className='plan-missing'>方案不存在</Text>
```

```tsx
// src/pages/catalog/detail.tsx
import { Image, Swiper, SwiperItem, Text, View } from '@tarojs/components'
import Taro, { useRouter } from '@tarojs/taro'
import { CATEGORY_LABELS, ROOM_LABELS, STYLE_LABELS } from '../../constants/labels'
import { useCatalogStore } from '../../stores/catalogStore'
import { useFavoriteStore } from '../../stores/favoriteStore'
import { usePlanStore } from '../../stores/planStore'
import { formatPriceYuan } from '../../utils/price'
import './detail.scss'

export default function CatalogDetailPage() {
  const router = useRouter()
  const productId = router.params.productId ?? ''
  const product = useCatalogStore((state) =>
    state.products.find((item) => item.id === productId),
  )
  const favorited = useFavoriteStore((state) =>
    state.items.some((item) => item.productId === productId),
  )
  const toggleFavorite = useFavoriteStore((state) => state.toggleFavorite)
  const activePlanId = usePlanStore((state) => state.activePlanId)
  const addProductToActivePlan = usePlanStore((state) => state.addProductToActivePlan)

  if (!product) {
    return (
      <View className='page'>
        <Text className='detail-missing'>商品不存在</Text>
      </View>
    )
  }

  const gallery = product.imageUrls.length > 0 ? product.imageUrls : [product.coverUrl]
  const roomText = product.rooms.map((room) => ROOM_LABELS[room]).join(' / ')

  const onAddToPlan = () => {
    if (activePlanId == null) {
      Taro.switchTab({ url: '/pages/plans/index' })
      return
    }
    const result = addProductToActivePlan(product.id)
    if (result === 'added') {
      Taro.showToast({ title: '已加入当前方案', icon: 'success' })
    } else if (result === 'duplicate') {
      Taro.showToast({ title: '已在当前方案中', icon: 'none' })
    }
  }

  return (
    <View className='page'>
      <Swiper className='detail-swiper' indicatorDots circular autoplay={false}>
        {gallery.map((src) => (
          <SwiperItem key={src}>
            <Image className='detail-swiper__image' src={src} mode='aspectFill' />
          </SwiperItem>
        ))}
      </Swiper>
      <Text className='detail-name'>{product.name}</Text>
      <Text className='detail-meta'>
        {CATEGORY_LABELS[product.category]} · {STYLE_LABELS[product.style]} · {roomText}
      </Text>
      <Text className='detail-price'>{formatPriceYuan(product.priceFen)}</Text>
      <Text className='detail-row'>
        尺寸 {product.sizeMm.w} × {product.sizeMm.d} × {product.sizeMm.h} mm
      </Text>
      <Text className='detail-row'>材质 {product.material}</Text>
      <Text className='detail-desc'>{product.description}</Text>
      <View className='detail-fav' onClick={() => toggleFavorite(product.id)}>
        <Text className='detail-fav__text'>{favorited ? '已收藏' : '收藏'}</Text>
      </View>
      <View
        className={activePlanId == null ? 'detail-plan detail-plan--ghost' : 'detail-plan'}
        onClick={onAddToPlan}
      >
        <Text
          className={
            activePlanId == null ? 'detail-plan__text detail-plan__text--ghost' : 'detail-plan__text'
          }
        >
          {activePlanId == null ? '请先新建方案' : '加入当前方案'}
        </Text>
      </View>
    </View>
  )
}
```

```ts
// src/app.ts
import { PropsWithChildren } from 'react'
import Taro from '@tarojs/taro'
import { useFavoriteStore } from './stores/favoriteStore'
import { usePlanStore } from './stores/planStore'
import { configureStorage } from './utils/storage'

import './app.scss'

configureStorage({
  getItem: (key) => {
    try {
      const value = Taro.getStorageSync(key)
      if (value === '' || value == null) return null
      return typeof value === 'string' ? value : JSON.stringify(value)
    } catch {
      return null
    }
  },
  setItem: (key, value) => {
    Taro.setStorageSync(key, value)
  },
})

useFavoriteStore.getState().hydrate()
usePlanStore.getState().hydrate()

function App({ children }: PropsWithChildren<any>) {
  return children
}

export default App
```
